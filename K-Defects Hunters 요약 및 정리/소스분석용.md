## 📌 전체 작동 원리 (요청 1건이 처리되는 흐름)

`router.py` 의 `send_chat_message()` 가 이 파일의 `chat()` 을 호출한다.  
`chat()` 안에서는 아래 순서로 **"위에서부터 걸러내는"** 구조로 동작한다.

1. **[1단계] 질문 분류**
   - `select_capabilities()` 로 이 질문이 어떤 컨텍스트(`docs`/`dashboard`/`inspection`)를 필요로 하는지 판단한다.
2. **[2단계] 지원하지 않는 질문인 경우**
   - 즉시 고정 답변 반환 (AI 호출 안 함)
3. **[3단계] "불량 PCB 몇 개?" 같은 수치 질문인 경우**
   - DB(대시보드 요약)에서 직접 계산해서 답변 (AI 호출 안 함)
4. **[4단계] 위 두 경우가 아닌 경우**
   - 실제 AI 호출
   - 컨텍스트 블록(문서/대시보드/배치 데이터)을 조립해서 시스템 프롬프트로 만들고, `provider.generate()` 로 AI에게 질문한다.

### 💡 핵심 포인트
- **모든 질문이 AI로 가는 게 아니다.** 2·3단계에서 걸리면 AI를 아예 안 부른다.
- **`provider`(Ollama/OpenAI)는 이 파일이 고르지 않는다.**
  - `dependencies.py` 의 `get_chat_provider()` 가 골라서 인자로 넣어준다.
  - 이 파일은 받은 `provider` 를 그냥 쓰기만 한다. (의존성 주입)

from collections.abc import Callable, Sequence

# 다른 기능(feature) 모듈의 조회 함수들을 "기본값"으로 가져온다.
# default_ 접두사가 붙은 이유: chat() 에서 인자로 다른 함수를 넣어주면
# 그걸 쓰고, 안 넣으면 이 기본 함수를 쓰기 때문. (테스트 시 교체하기 쉬움)
from ..batch.service import list_batches as default_list_batches
from ..dashboard.schemas import DashboardSummaryResponse
from ..dashboard.service import get_dashboard_summary as default_get_dashboard_summary

# capabilities.py: "이 질문이 어떤 종류인가"를 판단하는 규칙들이 모여 있는 모듈
from .capabilities import (
    UNSUPPORTED_OPERATIONAL_REPLY,   # 지원 안 하는 질문에 돌려줄 고정 문구
    is_defect_quantity_query,        # "불량 몇 개?" 류의 수치 질문인지 판별
    is_unsupported_operational_query,  # 지원 범위 밖 질문인지 판별
    select_capabilities,             # 질문 → 필요한 컨텍스트 목록으로 변환
)
from .knowledge import build_docs_context      # 프로젝트 문서 텍스트 블록 생성
from .prompts import build_system_prompt       # 시스템 프롬프트 조립
from .providers.base import ChatProvider, ProviderMessage  # provider 공통 인터페이스
from .schemas import ChatRequest, ChatResponse  # 요청/응답 데이터 형태 정의

# AI에게 넘길 때 Batch 목록을 최대 몇 건까지만 보여줄지 제한.
# 너무 많이 넣으면 토큰(비용)이 커지고 프롬프트가 지저분해지므로 5건으로 자름.
MAX_BATCH_CONTEXT_ROWS = 5

# 대시보드 데이터 조회 자체가 실패했을 때 사용자에게 보여줄 고정 문구.
DASHBOARD_LOAD_FAILURE_REPLY = (
    "현재 Dashboard 데이터를 조회할 수 없어 불량 PCB 수를 확인할 수 없습니다."
)


def select_contexts(message: str, context_hints: Sequence[str]) -> list[str]:
    """
    [기능] 사용자 질문 + 프론트가 준 힌트를 받아, 필요한 컨텍스트 목록을 돌려준다.
           (예: ["docs"], ["dashboard"], ["inspection"] 등)

    [참고] 실제 판단 로직은 capabilities.select_capabilities() 안에 있고,
           이 함수는 그걸 그대로 넘겨주는 얇은 래퍼(wrapper)일 뿐이다.
           chat() 내부에서는 이 함수 대신 select_capabilities() 를 직접 부르고 있어서,
           현재 이 함수는 외부(테스트 등)에서 쓰이는 용도로 남아 있는 상태다.
    """
    return select_capabilities(message, context_hints)


def build_dashboard_context(summary: DashboardSummaryResponse) -> str:
    """
    [기능] 대시보드 요약 데이터를 AI가 읽기 좋은 "텍스트 블록"으로 변환한다.

    [왜 필요한가] AI는 DB를 직접 못 본다. 그래서 숫자들을 사람이 읽는 형태의
                 문자열로 만들어 프롬프트에 끼워 넣어줘야 답변에 반영할 수 있다.

    [처리 순서]
      1) 불량 유형 중 count가 0보다 큰 것만 골라 상위 3개까지만 사용
      2) 하나도 없으면 "(집계된 불량 유형 없음)" 문구로 대체
      3) KPI 항목들을 줄 단위로 이어붙여 하나의 문자열로 반환
    """
    # count가 0인 유형은 의미가 없으므로 제외하고, 앞에서 3개만 사용([:3])
    top_defects = [
        item
        for item in summary.defect_types.items
        if item.count > 0
    ][:3]

    # 불량 유형이 있으면 항목별 줄을 만들고, 없으면 안내 문구 한 줄로 대체
    defect_lines = (
        [f"  - {item.type}: {item.count}건 ({item.rate}%)" for item in top_defects]
        if top_defects
        else ["  - (집계된 불량 유형 없음)"]
    )

    # 각 줄을 개행문자로 이어붙여 최종 블록 완성
    return "\n".join(
        [
            "[Dashboard 현재 데이터]",
            f"- 기준일: {summary.date.isoformat()}",
            f"- 오늘 검사 PCB 수: {summary.kpis.inspected_pcb_count}",
            f"- 불량 비율: {summary.kpis.defect_rate}%",
            f"- AI 판정 수용률: {summary.kpis.ai_acceptance_rate}%",
            f"- AI 검수 완료 객체 수: {summary.kpis.ai_reviewed_object_count}",
            f"- 검사 대기 Batch: {summary.batch_pipeline.waiting_inspection}",
            f"- 검사 중 Batch: {summary.batch_pipeline.inspecting}",
            f"- 검수 대기 Batch: {summary.batch_pipeline.waiting_review}",
            "- 주요 불량 유형:",
            *defect_lines,  # 리스트를 펼쳐서(unpack) 각 줄로 삽입
        ]
    )


def build_inspection_context(batches: Sequence[dict]) -> str:
    """
    [기능] Batch(검사 배치) 목록을 AI가 읽기 좋은 텍스트 블록으로 변환한다.

    [처리 순서]
      1) Batch가 하나도 없으면 → "등록된 Batch가 없습니다." 만 담아 즉시 반환
      2) 있으면 총 개수를 적고, 최대 MAX_BATCH_CONTEXT_ROWS(5)건까지만 상세 나열
      3) 각 Batch는 '코드 | 상태 | 라인 | 수량 | NG | 준비여부' 형태 한 줄로 요약
    """
    lines = ["[Inspection/Batch 현재 데이터]"]

    # 데이터가 없는 경우 조기 반환(early return) — 아래 로직을 탈 필요가 없음
    if not batches:
        lines.append("- 등록된 Batch가 없습니다.")
        return "\n".join(lines)

    lines.append(f"- 조회 Batch 수: {len(batches)}")
    lines.append("- 최근 Batch:")

    # 앞에서 5건만 잘라서 순회 (토큰 절약 목적)
    for batch in list(batches)[:MAX_BATCH_CONTEXT_ROWS]:
        # NG(불량) 개수가 None일 수 있으므로 '-'로 대체해 표시
        ng_count = batch.get("inspection_ng_count")
        ng_label = ng_count if ng_count is not None else "-"

        # True/False 값을 사람이 읽는 한글 라벨로 변환
        readiness = "준비 완료" if batch.get("inspection_ready") else "준비 미완료"

        # .get(키, 기본값) 을 쓰는 이유: 해당 키가 없어도 에러 없이 '-'로 표시하기 위함
        lines.append(
            "  - "
            f"{batch.get('batch_code', '-')}" | "
            f"상태={batch.get('status', '-')} | "
            f"라인={batch.get('line_name', '-')} | "
            f"수량={batch.get('batch_size', '-')} | "
            f"NG={ng_label} | "
            f"{readiness}"
        )
    return "\n".join(lines)


def defective_pcb_count_from_summary(summary: DashboardSummaryResponse) -> int:
    """
    [기능] 대시보드 요약값(검사 수 + 불량률)으로 실제 불량 PCB '개수'를 역산한다.

    [계산식] 불량 개수 = 검사 수 × (불량률 / 100), 반올림

    [주의] 검사 수가 0 이하이면 계산할 게 없으므로 0을 즉시 반환한다.
           (0으로 나누는 문제 방지 + 불필요한 계산 회피)
    """
    inspected = summary.kpis.inspected_pcb_count
    if inspected <= 0:
        return 0
    return round(inspected * summary.kpis.defect_rate / 100.0)


def build_defect_quantity_reply(summary: DashboardSummaryResponse) -> str:
    """
    [기능] "오늘 불량 PCB 몇 개야?" 류 질문에 대한 답변 문장을 직접 만든다.

    [중요] 이 답변은 AI가 만든 게 아니라, DB 숫자로 코드가 직접 조립한 문장이다.
           수치 질문은 AI에게 맡기면 틀릴 수 있으므로 일부러 AI를 거치지 않는다.
    """
    inspected = summary.kpis.inspected_pcb_count
    defective = defective_pcb_count_from_summary(summary)  # 위 함수로 개수 계산
    rate = summary.kpis.defect_rate
    date_label = summary.date.isoformat()
    reply = (
        f"오늘({date_label}) PCB 검사 기준 불량 PCB는 {defective}개입니다. "
        f"총 검사 PCB는 {inspected}개이며, 불량률은 {rate}%입니다."
    )
    return reply


def _build_context_blocks(
    contexts: Sequence[str],
    *,
    dashboard_loader: Callable[[], DashboardSummaryResponse],
    batch_loader: Callable[[], list[dict]],
) -> list[str]:
    """
    [기능] 필요한 컨텍스트 종류에 따라, AI 프롬프트에 넣을 텍스트 블록들을 모아 만든다.

    [이름 앞의 _ 의미] 이 모듈 내부에서만 쓰는 헬퍼 함수라는 관례적 표시.

    [인자 설명]
      contexts         : ["docs", "dashboard", "inspection"] 중 필요한 것들
      dashboard_loader : 대시보드 데이터를 가져오는 함수 (실제 DB 조회)
      batch_loader     : Batch 목록을 가져오는 함수 (실제 DB 조회)
      * 함수를 인자로 받는 이유: 테스트할 때 가짜 함수로 바꿔치기하기 위함(의존성 주입)

    [에러 처리 방식]
      DB 조회가 실패해도 try/except 로 잡아서 "조회할 수 없습니다" 문구를 대신 넣는다.
      → 대시보드 하나가 죽어도 챗봇 전체가 500 에러로 죽지 않고 답변은 나가게 하려는 설계.
    """
    blocks: list[str] = []

    # 1) 문서 컨텍스트: 정적 텍스트라 실패할 일이 없어 try/except 없음
    if "docs" in contexts:
        blocks.append(build_docs_context())

    # 2) 대시보드 컨텍스트: DB 조회라 실패 가능 → 실패 시 대체 문구
    if "dashboard" in contexts:
        try:
            blocks.append(build_dashboard_context(dashboard_loader()))
        except Exception:
            blocks.append(
                "[Dashboard 현재 데이터]\n현재 Dashboard 데이터를 조회할 수 없습니다."
            )

    # 3) 검사/Batch 컨텍스트: 마찬가지로 실패 시 대체 문구
    if "inspection" in contexts:
        try:
            blocks.append(build_inspection_context(batch_loader()))
        except Exception:
            blocks.append(
                "[Inspection/Batch 현재 데이터]\n현재 Batch 데이터를 조회할 수 없습니다."
            )
    return blocks


async def chat(
    request: ChatRequest,
    *,
    provider: ChatProvider,
    dashboard_loader: Callable[[], DashboardSummaryResponse] | None = None,
    batch_loader: Callable[[], list[dict]] | None = None,
) -> ChatResponse:
    """
    [기능] 이 파일의 메인 함수. 챗봇 요청 1건을 처음부터 끝까지 처리한다.

    [인자 설명]
      request          : 사용자 질문(message)과 힌트(context_hints)를 담은 객체
      provider         : 실제 AI 호출을 담당하는 객체 (OllamaProvider 또는 OpenAIProvider)
                         ★ 이 함수는 provider 를 고르지 않는다. router → dependencies 에서
                           이미 결정되어 인자로 들어온다.
      dashboard_loader : (선택) 대시보드 조회 함수. 안 주면 실제 DB 함수 사용
      batch_loader     : (선택) Batch 조회 함수. 안 주면 실제 DB 함수 사용

    [* 의 의미] * 뒤의 인자들은 반드시 이름을 붙여 호출해야 한다(키워드 전용 인자).
                예: chat(request, provider=my_provider)

    [async 인 이유] AI 호출은 네트워크 대기 시간이 길어서, 기다리는 동안
                    다른 요청을 처리할 수 있도록 비동기로 작성됨.
    """
    # ── [1단계] 질문 분류: 어떤 컨텍스트가 필요한지 판단 ──────────────
    contexts = select_capabilities(request.message, request.context_hints)

    # ── [2단계] 지원 범위 밖 질문이면 AI를 부르지 않고 고정 답변 반환 ──
    #   예: 챗봇이 다루기로 한 영역이 아닌 운영 관련 질문 등
    if is_unsupported_operational_query(request.message, contexts):
        return ChatResponse(
            reply=UNSUPPORTED_OPERATIONAL_REPLY,
            contexts_used=[],       # AI도 컨텍스트도 안 썼으므로 빈 목록
            provider="ollama",      # ※ 하드코딩 (아래 맨 끝 주석 참고)
        )

    # 호출자가 로더 함수를 안 넘겼으면 실제 DB 조회 함수를 기본값으로 사용
    # (or 연산: 앞이 None이면 뒤 값을 쓴다)
    load_dashboard = dashboard_loader or default_get_dashboard_summary
    load_batches = batch_loader or default_list_batches

    # ── [3단계] 수치 질문("불량 몇 개?")이면 DB 숫자로 직접 답변 ────────
    #   AI에게 맡기면 숫자를 지어낼 위험이 있어 코드가 직접 계산한다.
    if is_defect_quantity_query(request.message):
        try:
            summary = load_dashboard()
        except Exception:
            # 대시보드 조회 실패 → 계산이 불가능하므로 실패 안내 문구 반환
            return ChatResponse(
                reply=DASHBOARD_LOAD_FAILURE_REPLY,
                contexts_used=["dashboard"],
                provider="ollama",   # ※ 하드코딩
            )
        return ChatResponse(
            reply=build_defect_quantity_reply(summary),  # 코드가 만든 문장
            contexts_used=["dashboard"],
            provider="ollama",       # ※ 하드코딩
        )

    # ── [4단계] 위에 안 걸린 일반 질문 → 실제 AI 호출 ────────────────

    # 4-1) 필요한 컨텍스트 블록들(문서/대시보드/Batch)을 모아 온다
    context_blocks = _build_context_blocks(
        contexts,
        dashboard_loader=load_dashboard,
        batch_loader=load_batches,
    )

    # 4-2) AI에게 보낼 메시지 2개를 구성
    #       system : 역할 지시 + 위에서 만든 실제 데이터 컨텍스트
    #       user   : 사용자가 실제로 물어본 질문
    messages = [
        ProviderMessage(role="system", content=build_system_prompt(context_blocks)),
        ProviderMessage(role="user", content=request.message),
    ]

    # 4-3) 실제 AI 호출 지점.
    #       provider 가 OllamaProvider면 로컬 Ollama로,
    #       OpenAIProvider면 OpenAI API로 나간다. 이 코드는 둘을 구분하지 않는다.
    #       (await = 응답이 올 때까지 기다림)
    reply = await provider.generate(messages)

    # 4-4) 최종 응답 반환
    #
    # ⚠️⚠️ 여기가 지금 문제가 된 지점 ⚠️⚠️
    #   provider="ollama" 가 문자열로 고정되어 있다.
    #   즉 실제로는 OpenAI가 답변했더라도 응답 JSON에는 항상 "ollama"로 표시된다.
    #   → 응답의 provider 필드는 "어느 AI가 답했는지"를 알려주는 신뢰할 수 있는
    #      지표가 아니다. (실제 동작에는 영향 없음. 표시값만 틀림)
